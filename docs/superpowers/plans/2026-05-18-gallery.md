# Gallery Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Ship a `/[lang]/gallery/` page that displays photography work in a Flickr/500px-style justified-rows layout, with a click-to-open lightbox, tag-chip filtering, and full Astro Image optimization.

**Architecture:** Single Astro page composed from small `.astro` components — Hero, TagFilter, Grid, Item, Lightbox. Photos come from a new `gallery` content collection, image binaries from `src/assets/gallery/`. Layout is CSS-only (flex justified rows), filtering is CSS-only (attribute selectors), lightbox is vanilla JS with URL-hash sync.

**Tech Stack:** Astro 6.3, TypeScript, Tailwind v4, Astro `<Image>` (Sharp), shadcn neutral palette, Iconify `lucide:` icons.

**Reference spec:** `docs/superpowers/specs/2026-05-18-gallery-design.md`

**Verification model for this codebase:** There is no unit-test framework. Each task "test" step uses `pnpm build` (full TS + render check) and, for interactive features, a manual smoke check via `pnpm dev`. Lefthook runs biome on staged files automatically on commit; biome skips `.astro` files (per `biome.jsonc`).

Create `src/lib/photos.ts` with:

```ts
import type { ImageMetadata } from 'astro';

// Eagerly import every gallery photo. Astro processes each (content-hashed
// URL, intrinsic width/height parsed). Look up by basename so frontmatter
// stays simple (`image: foo.jpg`).
const photos = import.meta.glob<{ default: ImageMetadata }>(
  '/src/assets/gallery/*.{webp,png,jpg,jpeg,avif}',
  { eager: true },
);

export function resolveGalleryImage(name: string): ImageMetadata {
  const key = `/src/assets/gallery/${name}`;
  const mod = photos[key];
  if (!mod) {
    const available = Object.keys(photos)
      .map((k) => k.replace('/src/assets/gallery/', ''))
      .join(', ');
    throw new Error(
      `Gallery image not found: "${name}". Place it in src/assets/gallery/. Available: ${available}`,
    );
  }
  return mod.default;
}
```

import type { CollectionEntry } from "astro:content";
import { t } from "../../i18n/ui";
import { pluralKey } from "../../i18n/utils";
import type { Lang } from "../../i18n/config";

interface Props {
  lang: Lang;
  photos: CollectionEntry<"gallery">[];

const { lang, photos } = Astro.props;

const count = photos.length;
const countKey = pluralKey("gallery.count", count);

const sorted = [...photos].sort((a, b) => +a.data.date - +b.data.date);
const earliest = sorted[0]?.data.date;
const latest = sorted[sorted.length - 1]?.data.date;

let dateRange = "";
if (earliest && latest) {
  const sy = earliest.getFullYear();
  const ey = latest.getFullYear();
  dateRange =
    sy === ey
      ? t(lang, "gallery.dateRangeSingle", { year: sy })
      : t(lang, "gallery.dateRangeSpan", { start: sy, end: ey });
}
---

  <h1 class="text-4xl font-semibold tracking-tight text-foreground lg:text-5xl">
    {t(lang, "gallery.title")}
  </h1>
  <p class="mt-4 max-w-prose text-lg text-muted-foreground">
    {t(lang, "gallery.subtitle")}
  </p>
  {
    count > 0 && (
      <p class="mt-6 text-sm text-muted-foreground">
        <span>{t(lang, countKey, { n: count })}</span>
        {dateRange && (
          <>
            <span aria-hidden="true"> · </span>
            <span>{dateRange}</span>
          </>

  // Filter items whose data-tags contains the active tag. Pure DOM, no framework.
  (function () {
    function init() {
      const grid = document.getElementById("gallery-grid");
      const empty = document.getElementById("gallery-empty");
      if (!grid) return;

      function apply() {
        const tag = grid.getAttribute("data-active-tag") || "all";
        const items = Array.from(grid.querySelectorAll(".gallery-item"));
        let visible = 0;
        for (const el of items) {
          const tags = (el.getAttribute("data-tags") || "").split(/\s+/);
          const show = tag === "all" || tags.includes(tag);
          el.style.display = show ? "" : "none";
          if (show) visible++;
        }
        if (empty) empty.classList.toggle("hidden", visible > 0);
      }

      grid.addEventListener("gallery:tag-changed", apply);
      apply();
    }

    if (document.readyState === "loading") {
      document.addEventListener("DOMContentLoaded", init, { once: true });
    } else {
      init();
    }
  })();
</script>
```

Note: the CSS `[data-active-tag]:not([data-active-tag="all"])` selector exists as a hint, but actual filtering is handled by the inline script (which sets `style.display`). This avoids brittleness with attribute-value escaping and lets us also count visible items for the empty state.

      return Array.from(grid.querySelectorAll(".gallery-item")).filter(
        (el) => el.offsetParent !== null,
      );
    }

    function renderSlug(slug) {
      const tpl = document.getElementById("photo-" + slug);
      if (!(tpl instanceof HTMLTemplateElement)) return false;
      const clone = tpl.content.cloneNode(true);
      const photoEl = clone.querySelector(".lb-photo");
      const metaEl = clone.querySelector(".lb-meta");
      photoSlot.replaceChildren(photoEl);
      metaSlot.replaceChildren(metaEl);
      return true;
    }

    function open(slug) {
      if (!renderSlug(slug)) return;
      currentSlug = slug;
      dialog.classList.remove("hidden");
      if (typeof dialog.showModal === "function" && !dialog.open) {
        try { dialog.showModal(); } catch {}
      }
      if (location.hash !== "#" + slug) {
        history.replaceState(null, "", "#" + slug);
      }
      document.body.style.overflow = "hidden";
    }

    function close() {
      dialog.classList.add("hidden");
      if (typeof dialog.close === "function" && dialog.open) {
        try { dialog.close(); } catch {}
      }
      photoSlot.replaceChildren();
      metaSlot.replaceChildren();
      currentSlug = null;
      if (location.hash) {
        history.replaceState(null, "", location.pathname + location.search);
      }
      document.body.style.overflow = "";
    }

    function step(delta) {
      if (!currentSlug) return;
      const items = visibleItems();
      if (items.length === 0) return;
      const idx = items.findIndex((el) => el.getAttribute("data-slug") === currentSlug);
      if (idx === -1) {
        open(items[0].getAttribute("data-slug"));
        return;
      }
      const next = items[(idx + delta + items.length) % items.length];
      const slug = next.getAttribute("data-slug");
      if (slug) open(slug);
    }

    grid.addEventListener("click", (e) => {
      const target = e.target;
      if (!(target instanceof Element)) return;
      const item = target.closest(".gallery-item");
      if (!item) return;
      const slug = item.getAttribute("data-slug");
      if (slug) open(slug);
    });

    closeBtn?.addEventListener("click", close);
    prevBtn?.addEventListener("click", () => step(-1));
    nextBtn?.addEventListener("click", () => step(1));

    dialog.addEventListener("click", (e) => {
      // Click on backdrop (dialog itself, not inner content) closes the lightbox.
      if (e.target === dialog) close();
    });

    document.addEventListener("keydown", (e) => {
      if (dialog.classList.contains("hidden")) return;
      if (e.key === "Escape") close();
      else if (e.key === "ArrowLeft") step(-1);
      else if (e.key === "ArrowRight") step(1);
    });

    window.addEventListener("hashchange", () => {
      const slug = location.hash.replace(/^#/, "");
      if (!slug) {
        if (!dialog.classList.contains("hidden")) close();
        return;
      }
      if (slug !== currentSlug) open(slug);
    });

    // Deep link on initial load
    const initial = location.hash.replace(/^#/, "");
    if (initial) open(initial);
    }

    if (document.readyState === "loading") {
      document.addEventListener("DOMContentLoaded", init, { once: true });
    } else {
      init();
    }
  })();
</script>
```
